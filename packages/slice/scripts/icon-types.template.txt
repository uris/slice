import type React from 'react';
import type { ColorValue } from '../../theme/colors/resolveColor';

type IconBaseProps = {
	name?: SliceIcon;
	size?: number;
	stroke?: number;
	strokeColor?: ColorValue;
	fillOpacity?: number;
	iconName?: string;
	actionName?: string;
	decorative?: boolean;
	coverUp?: ColorValue;
	toggle?: boolean;
	pointer?: boolean;
	disabled?: boolean;
	fill?: boolean;
	onClick?: (e: React.MouseEvent<SVGElement, MouseEvent>) => void;
};

export type IconProps = Omit<React.SVGAttributes<SVGSVGElement>, keyof IconBaseProps | keyof React.AriaAttributes | 'role'> & IconBaseProps;
