        # useIntersecting
        Tracks intersection of one or more elements — by ref or CSS selector — against a container using `IntersectionObserver`, returning the latest intersection state for each resolved target. Commonly used for infinite scroll sentinels, lazy loading, and view/impression tracking.

        String selectors are resolved via `document.querySelectorAll` whenever the resolved target list actually changes; prefer refs for elements that mount after the initial render, or for one-of-many list items.

        ## Signature
        `useIntersecting(options?)`

        ## Code Examples

        <RenderSource language={"tsx"}>
        ```
        import { useRef } from 'react';
        import { defaultOptions, useIntersecting } from '@apple-pie/slice';

        function InfiniteList() {
            const containerRef = useRef<HTMLDivElement>(null);
            const sentinelRef = useRef<HTMLDivElement>(null);

            const [sentinel] = useIntersecting({
                ...defaultOptions,
                container: containerRef,
                entries: sentinelRef,
                thresholds: 0.1,
                margin: 200,
            });

            useEffect(() => {
                if (sentinel?.isIntersecting) loadNextPage();
            }, [sentinel?.isIntersecting]);

            return (
                <div ref={containerRef} style={{ overflowY: 'auto', height: 320 }}>
                    {items.map((item) => <Row key={item.id} item={item} />)}
                    <div ref={sentinelRef} />
                </div>
            );
        }
        ```
